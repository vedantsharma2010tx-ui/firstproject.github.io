    function checkZip() {

      const result = document.getElementById('zip-result');
      result.classList.remove('hidden', 'text-green-300', 'text-amber-300');
      if (/^\d{5}$/.test(zip)) {
        result.classList.add('text-green-300');
        result.innerHTML = 'Great news! We have 2 licensed plumbers on standby near <strong>' + zip + '</strong>.';
      } else {
        result.classList.add('text-amber-300');
        result.textContent = 'Please enter a valid 5-digit ZIP code.';
      }
    }

    function filterServices(category) {
      const cards = document.querySelectorAll('.service-card');
      const tabs = document.querySelectorAll('.service-tab');

      tabs.forEach(function(tab) {
        if (tab.dataset.category === category) {
          tab.className = 'service-tab bg-sky-900 text-white px-5 py-2 rounded-full text-sm font-semibold transition';
        } else {
          tab.className = 'service-tab bg-slate-200 hover:bg-slate-300 text-slate-700 px-5 py-2 rounded-full text-sm font-semibold transition';
        }
      });

      cards.forEach(function(card) {
        if (category === 'all' || card.dataset.category === category) {
          card.classList.remove('hidden');
        } else {
          card.classList.add('hidden');
        }
      });
    }

    function calculateEstimate() {
      const rangeStr = document.getElementById('est-service').value;
      const surcharge = parseInt(document.getElementById('est-timing').value, 10);
      const parts = rangeStr.split('-').map(Number);
      const min = parts[0] + surcharge;
      const max = parts[1] + surcharge;
      document.getElementById('est-output').textContent = '$' + min.toLocaleString() + ' - $' + max.toLocaleString();
    }

    }
